// ✨ ฟังก์ชันแปลงลิงก์ Google Drive ให้ใช้งานกับ AR ได้ทันที function handleDriveLink(i, driveUrl) { if (!driveUrl.includes("drive.google.com")) return; try { // ดึง File ID ออกจากลิงก์ Google Drive const parts = driveUrl.split('/d/'); if (parts.length > 1) { const fileId = parts[1].split('/')[0]; // สร้างลิงก์ Direct Download ที่ Google Sheets และ AR ยอมรับ const directLink = `https://docs.google.com/uc?export=download&id=${fileId}`; // ใส่ค่ากลับไปที่ช่อง URL และสั่ง Preview document.getElementById(`m-${i}`).value = directLink; preview(i); // แจ้งเตือนคุณครูให้สบายใจ console.log("แปลงลิงก์ Google Drive สำเร็จ!"); } } catch (e) { alert("ลิงก์ Google Drive ไม่ถูกต้อง โปรดตรวจสอบการแชร์ไฟล์ครับ"); } }
